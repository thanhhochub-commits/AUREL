(function(){
  function syncRange(el){
    if(!el||el.type!=='range') return;
    var min=parseFloat(el.min);
    var max=parseFloat(el.max);
    var val=parseFloat(el.value);
    if(!isFinite(min)) min=0;
    if(!isFinite(max)) max=100;
    if(!isFinite(val)) val=min;
    var pct=max===min?0:((val-min)/(max-min))*100;
    pct=Math.max(0,Math.min(100,pct));
    el.style.setProperty('--range-pct',pct+'%');
  }

  function initScenarioRanges(){
    document.querySelectorAll('input[type="range"]').forEach(function(el){
      syncRange(el);
      if(el.dataset.aurelRangeReady==='1') return;
      el.dataset.aurelRangeReady='1';
      el.addEventListener('input',function(){syncRange(el);});
      el.addEventListener('change',function(){syncRange(el);});
    });
  }

  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded',initScenarioRanges,{once:true});
  }else{
    initScenarioRanges();
  }

  var root=document.getElementById('root')||document.body;
  if(root){
    new MutationObserver(function(){initScenarioRanges();})
      .observe(root,{childList:true,subtree:true});
  }
})();

<script id="AUREL_R128_RANGE_SYNC">
(function(){
  function syncRange(el){
    if(!el||el.type!=='range') return;
    var min=parseFloat(el.min), max=parseFloat(el.max), val=parseFloat(el.value);
    if(!isFinite(min)) min=0;
    if(!isFinite(max)) max=100;
    if(!isFinite(val)) val=min;
    var pct=max===min?0:((val-min)/(max-min))*100;
    pct=Math.max(0,Math.min(100,pct));
    el.style.setProperty('--range-pct',pct+'%');
  }
  function init(){
    document.querySelectorAll('input[type="range"]').forEach(function(el){
      syncRange(el);
      if(el.dataset.aurelR128==='1') return;
      el.dataset.aurelR128='1';
      el.addEventListener('input',function(){syncRange(el);});
      el.addEventListener('change',function(){syncRange(el);});
    });
  }
  if(document.readyState==='loading') document.addEventListener('DOMContentLoaded',init,{once:true});
  else init();
  var root=document.getElementById('root')||document.body;
  if(root) new MutationObserver(init).observe(root,{childList:true,subtree:true});
})();
</script>
